4. Add manual J&T shipment tables if missing.

Create shipments table if missing:

CREATE TABLE IF NOT EXISTS shipments (
  id BIGSERIAL PRIMARY KEY,
  order_id BIGINT NOT NULL REFERENCES orders(id),
  provider TEXT NOT NULL DEFAULT 'manual',
  courier TEXT NOT NULL DEFAULT 'jnt',
  courier_name TEXT DEFAULT 'J&T Express',
  service_type TEXT DEFAULT 'standard',
  waybill_number TEXT NOT NULL,
  tracking_number TEXT NOT NULL,
  shipping_fee NUMERIC(12,2) DEFAULT 0,
  status TEXT NOT NULL DEFAULT 'waybill_created',
  label_url TEXT,
  metadata JSONB DEFAULT '{}'::jsonb,
  created_by BIGINT,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE UNIQUE INDEX IF NOT EXISTS idx_shipments_tracking_number
ON shipments(tracking_number);

CREATE UNIQUE INDEX IF NOT EXISTS idx_shipments_waybill_number
ON shipments(waybill_number);

CREATE UNIQUE INDEX IF NOT EXISTS idx_shipments_active_order
ON shipments(order_id)
WHERE status NOT IN ('cancelled', 'returned');

Create shipment_events table if missing:

CREATE TABLE IF NOT EXISTS shipment_events (
  id BIGSERIAL PRIMARY KEY,
  shipment_id BIGINT NOT NULL REFERENCES shipments(id),
  provider TEXT DEFAULT 'manual',
  status TEXT NOT NULL,
  description TEXT,
  location TEXT,
  event_time TIMESTAMPTZ DEFAULT NOW(),
  raw_event JSONB DEFAULT '{}'::jsonb,
  created_at TIMESTAMPTZ DEFAULT NOW()
